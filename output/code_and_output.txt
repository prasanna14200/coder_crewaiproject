import math

# Define the function to calculate the series
def calculate_series(n):
    total = 0
    for i in range(1, n + 1):
        # Check if i is odd or even
        if i % 2 != 0:
            # If i is odd, add 1 / i to the total
            total += 1 / i
        else:
            # If i is even, subtract 1 / i from the total
            total -= 1 / i
    return total * 4

# Calculate the series for the first 10,000 terms
result = calculate_series(10000)
print(result)
